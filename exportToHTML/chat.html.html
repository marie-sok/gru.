<script>

    let currentChat = null;
    let messages = [];

    function openChat(id) {
        currentChat = id;
        document.getElementById("chatName").innerText = "Chat " + id;
        loadMessages();
    }

    function loadMessages() {
        let container = document.getElementById("messages");
        container.innerHTML = "";

        messages.forEach(m => {
            if (m.chat === currentChat) {
                let div = document.createElement("div");
                div.className = "msg " + (m.me ? "me" : "them");
                div.innerText = m.text + (m.read ? " ✔✔" : " ✔");

                div.onclick = () => deleteMsg(m.id);

                container.appendChild(div);
            }
        });
    }

    function send() {
        let text = document.getElementById("msg").value;
        let file = document.getElementById("fileInput").files[0];

        let msg = {
            id: Date.now(),
            chat: currentChat,
            text: text || (file ? "📎 " + file.name : ""),
            me: true,
            read: false
        };

        messages.push(msg);
        document.getElementById("msg").value = "";

        loadMessages();

        // имитация ответа
        setTimeout(() => {
            messages.push({
                id: Date.now()+1,
                chat: currentChat,
                text: "Ответ",
                me: false,
                read: true
            });
            loadMessages();
        }, 1000);
    }

    function deleteMsg(id) {
        messages = messages.filter(m => m.id !== id);
        loadMessages();
    }

    /* THEME */
    function toggleTheme() {
        let body = document.body;
        body.className = body.className === "light" ? "dark" : "light";
    }

    /* SETTINGS */
    function openSettings() {
        document.getElementById("settings").style.display = "block";
    }

    function closeSettings() {
        document.getElementById("settings").style.display = "none";
    }

    function addToBlacklist() {
        alert("User added to blacklist");
    }

</script>
