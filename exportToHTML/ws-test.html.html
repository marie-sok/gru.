<!DOCTYPE html>
<html>
<head>
    <script src="https://cdn.jsdelivr.net/npm/sockjs-client/dist/sockjs.min.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/stompjs/lib/stomp.min.js"></script>
</head>

<body>

<input id="msg" placeholder="message">
<button onclick="send()">Send</button>

<script>
    let socket = new SockJS('http://localhost:8081/ws');
    let stomp = Stomp.over(socket);

    let TOKEN = "functional_unique_create_kerry_wtf_its_a_rofl??";

    stomp.connect(
      {Authorization: "Bearer " + TOKEN},
      function(frame) {

        stomp.subscribe('/topic/chat/1', function(msg) {
            console.log("Получено:", msg.body);
        });

      }
    );

    function send() {
        stomp.send("/app/chat.send", {}, JSON.stringify({
            senderId: 1,
            receiverId: 1,
            content: document.getElementById("msg").value
        }));
    }
</script>

</body>
</html>
